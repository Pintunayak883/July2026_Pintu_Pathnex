resource "aws_security_group" "web_sg" {
    name           = "web-security-group"
    description    = "Allow SSH and HTTP"


    ingress {
        description  = "SSH Access"
        form_port    = 22
        to_port      = 22
        protocol     = "tcp"
        cidr_blocks  = ["0.0.0.0/0"]
    }

    ingress {
        description  = "HTTP Access"
        form_port    = 80
        to_port      = 80
        protocol     = "tcp"
        cidr_blocks  = ["0.0.0.0/0"]
    }

    egress {
        description  = "Allow All Outbound Traffic"
        form_port    = 0
        to_port      = 0
        protocol     = "-1"
        cidr_blocks  = ["0.0.0.0/0"]
    }

    tags = {
        Name = "Terraform-SG"
    }
}

## Jenkins Pipeline
You will learn to **combine all tasks into a realistic Jenkins Pipeline** with proper stages and tags.

pipeline {
    agent any

    environment {
        INSTITUTE_NAME = "Pathnex"
        TEAM           = "DevOps"
        ENV            = "prod"
        PROJECT        = "Pathnex-Training"
    }

    stages {
        stage('Infrastructure') {
            steps {
                steps {
                    echo "Creating AWS Infrastructure..."

                    sh 'terraform init'
                    sh 'terraform validate'
                    sh 'terraform plan'
                    sh 'terraform apply -auto-approve'
                }
            }
            stage('App Deployment') {
                steps {
                    echo "Deploying Application..."

                    sh 'ansible-playbook playbook.yml'

                    sh 'kubectl apply -f k8s.yml'
                }
            }
            stage('Monitoring Checks') {
                steps {
                    echo "Running System Health Checks..."
                    
                    sh 'chmod +x system-check.sh'

                    sh './system-check.sh'
                }
            }
        }
        post {

            success {
                echo "Pipeline Completed Successfully"
            }
            failure {
                echo "Pipeline Failed"
            }
            always {
                echo "Pipeline Execution Finished"
            }
        }
    }
}
